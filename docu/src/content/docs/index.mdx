import { Card, CardGrid } from '@astrojs/starlight/components';

## Módulos Disponibles

<CardGrid stagger>
	<Card title="Usuarios" icon="user">
		Gestión completa de usuarios, incluyendo autenticación, roles, permisos y perfiles. Soporta creación, actualización, eliminación y exportación de usuarios.
	</Card>
	<Card title="Cultivos" icon="leaf">
		Administración de cultivos agrícolas, con seguimiento de cantidades, imágenes, tipos y fechas de plantado. Incluye reportes Excel detallados.
	</Card>
	<Card title="Lotes" icon="map">
		Gestión de lotes de terreno con coordenadas geoespaciales, áreas, estados y estadísticas. Soporta cache para optimización de consultas.
	</Card>
	<Card title="Materiales" icon="tool">
		Inventario de materiales agrícolas con categorización jerárquica, stock bajo, precios y proveedores. Incluye gestión de imágenes y soft delete.
	</Card>
	<Card title="Producciones" icon="chart">
		Seguimiento de producciones por cultivo, con estadísticas de ingresos/gastos, estados de producción y disponibilidad para venta.
	</Card>
	<Card title="Ventas" icon="shopping-cart">
		Registro de ventas con actualización automática de inventario, generación de facturas PDF y reportes de flujo mensual.
	</Card>
</CardGrid>
